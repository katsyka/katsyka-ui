th.innerHTML='<tr>'+COLS.map(([k,n])=>`<th ${k===key?`aria-sort="${dir>0?'ascending':'descending'}"`:''}><button data-k="${k}">${n}@I(up)</button></th>`).join('')+'<th><span class="vh">Actions</span></th></tr>';
tb.innerHTML=L.slice(pg*sz,pg*sz+sz).map(u=>`<tr class="${u.s==='suspended'?'sus':''}"><td><span class="u"><span class="av" style="--c:${u.c}">${u.n.split(' ').map(w=>w[0]).join('')}</span><div><b>${u.n}</b><small>${u.e}</small></div></span></td><td><select data-r="${u.e}" aria-label="Role for ${u.n}" ${u.r==='Owner'?'disabled':''}>${['Owner','Admin','Member','Viewer'].map(r=>`<option ${r===u.r?'selected':''} ${r==='Owner'&&u.r!=='Owner'?'disabled':''}>${r}</option>`).join('')}</select></td><td><span class="pill ${u.s==='active'?'ok':u.s==='invited'?'info':'err'}">${u.s[0].toUpperCase()+u.s.slice(1)}</span></td><td class="mu">${ago(u.l)}</td><td style="text-align:right">${u.s==='invited'?`<button class="b gh sm" data-rs="${u.e}">Resend invite</button>`:u.r!=='Owner'?`<button class="b gh sm" data-su="${u.e}">${u.s==='suspended'?'Restore':'Suspend'}</button>`:''}</td></tr>`).join('');
document.getElementById('pi').textContent=`${pg*sz+1}–${Math.min(L.length,pg*sz+sz)} of ${L.length}`;document.getElementById('pp').disabled=pg===0;document.getElementById('np').disabled=pg>=np-1;document.getElementById('ct').textContent=U.length}
th.onclick=e=>{const b=e.target.closest('[data-k]');if(!b)return;dir=key===b.dataset.k?-dir:1;key=b.dataset.k;draw();th.querySelector(`[data-k="${key}"]`).focus()};
tb.onchange=e=>{const u=U.find(x=>x.e===e.target.dataset.r);if(u){const o=u.r;u.r=e.target.value;toast(`${u.n} is now ${u.r}`,()=>{u.r=o})}};
tb.onclick=e=>{const s=e.target.closest('[data-su]');if(s){const u=U.find(x=>x.e===s.dataset.su),o=u.s;u.s=o==='suspended'?'active':'suspended';draw();toast(`${u.n} ${u.s==='suspended'?'suspended':'restored'}`,()=>{u.s=o})}const r=e.target.closest('[data-rs]');if(r){r.textContent='Sent';r.disabled=true}};
let tt;function toast(m,undo){last=undo;const t=document.getElementById('ts');document.getElementById('tt').textContent=m;t.hidden=false;clearTimeout(tt);tt=setTimeout(()=>t.hidden=true,5000);draw()}
document.getElementById('un').onclick=()=>{if(last)last();last=null;document.getElementById('ts').hidden=true;draw()};
document.getElementById('pp').onclick=()=>{pg--;draw()};document.getElementById('np').onclick=()=>{pg++;draw()};ps.onchange=()=>{pg=0;draw()};
document.getElementById('csv').onclick=()=>{const a=document.createElement('a');a.href=URL.createObjectURL(new Blob(['name,email,role,status\n'+U.map(u=>[u.n,u.e,u.r,u.s].join(',')).join('\n')],{type:'text/csv'}));a.download='users.csv';a.click()};draw();
</script>
